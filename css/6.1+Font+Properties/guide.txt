┌─────────────────────────────────────────────────────────────┐
│  LEVEL 1: ESSENTIALS (Master These First)                  │
└─────────────────────────────────────────────────────────────┘

These 5 properties are used on EVERY project.

body {
  /* 1. FONT FAMILY - Always provide fallbacks */
  font-family: "Inter", "Segoe UI", Roboto, "Helvetica Neue", Arial, sans-serif;
  
  /* System font stack (ultra-safe): */
  font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, 
               Oxygen, Ubuntu, Cantarell, sans-serif;

  /* 2. FONT SIZE - Use rem for accessibility */
  font-size: 1rem;        /* 16px default - PREFERRED */
  font-size: 16px;        /* Only use px for fixed elements */

  /* 3. FONT WEIGHT - Numeric is more consistent */
  font-weight: 400;       /* normal */
  font-weight: 700;       /* bold */
  /* Values: 100(thin), 200, 300(light), 400(normal), 500(medium), 
     600(semi), 700(bold), 800, 900(black) */

  /* 4. LINE HEIGHT - Improves readability */
  line-height: 1.5;       /* Unitless = relative to font-size (best) */
  line-height: 1.6;       /* For body text, 1.5–1.7 is ideal */

  /* 5. COLOR */
  color: #1a1a1a;         /* Dark gray > pure black (#000) for readability */
}

┌─────────────────────────────────────────────────────────────┐
│  LEVEL 2: INTERMEDIATE (Polish & Refinement)               │
└─────────────────────────────────────────────────────────────┘

p {
  /* TEXT ALIGNMENT */
  text-align: left;       /* left, right, center, justify */
  text-align: justify;    /* Clean edges - use sparingly */

  /* LETTER SPACING (Tracking) - subtle improvements */
  letter-spacing: 0.02em;  /* Normal text */
  letter-spacing: 0.05em;  /* Uppercase text (improves readability) */
  letter-spacing: -0.01em; /* Large headings (tightens them) */

  /* WORD SPACING - rarely used but useful */
  word-spacing: 0.1em;

  /* TEXT TRANSFORM */
  text-transform: uppercase;    /* ALL CAPS */
  text-transform: capitalize;   /* First Letter Of Each Word */
  text-transform: lowercase;    /* all lowercase */

  /* TEXT DECORATION */
  text-decoration: none;         /* Remove underlines from links */
  text-decoration: underline;    /* Standard underline */
  text-decoration: underline wavy #ff6b6b 2px; /* Custom underline! */

  /* FONT STYLE */
  font-style: italic;        /* For emphasis */
  font-style: normal;

  /* FONT VARIANT - small caps (classy!) */
  font-variant: small-caps;
}

--- PRO TIP: Letter Spacing Combinations ---

h1 {
  font-size: 2.5rem;
  letter-spacing: -0.02em;    /* Tighter for big text */
}

.small-uppercase {
  text-transform: uppercase;
  letter-spacing: 0.08em;     /* Wider for uppercase small text */
  font-size: 0.75rem;
}

┌─────────────────────────────────────────────────────────────┐
│  LEVEL 3: ADVANCED (Professional Techniques)               │
└─────────────────────────────────────────────────────────────┘

.advanced-text {
  /* SHORTHAND: font (order matters!) */
  /* Order: style | variant | weight | size/line-height | family */
  font: italic small-caps 700 1.125rem/1.7 "Inter", sans-serif;

  /* TEXT SHADOW (3D depth & readability) */
  /* Soft shadow (dark text on light bg): */
  text-shadow: 0 1px 2px rgba(0, 0, 0, 0.05);

  /* Glow effect (light text on dark bg): */
  text-shadow: 0 0 10px rgba(255, 255, 255, 0.5);

  /* 3D effect: */
  text-shadow: 0 2px 0 #ccc, 0 4px 0 #aaa, 0 6px 0 #888;

  /* TEXT OVERFLOW (long text handling) */
  white-space: nowrap;
  overflow: hidden;
  text-overflow: ellipsis;    /* ... at the end */

  /* WORD BREAK (prevent layout breaks) */
  word-break: break-word;     /* Breaks long words */
  overflow-wrap: break-word;  /* Same but more standard */

  /* HYPHENS (print-quality text) */
  hyphens: auto;              /* Adds hyphens at line breaks */
  -webkit-hyphens: auto;

  /* FONT SMOOTHING (macOS/Windows) - use with caution */
  -webkit-font-smoothing: antialiased;  /* macOS */
  -moz-osx-font-smoothing: grayscale;   /* macOS Firefox */
}

┌─────────────────────────────────────────────────────────────┐
│  SYSTEM FONT STACKS (Copy-Paste Ready)                     │
└─────────────────────────────────────────────────────────────┘

/* OPTION 1: Modern System Font Stack (cleanest) */
body {
  font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, 
               Oxygen-Sans, Ubuntu, Cantarell, "Helvetica Neue", sans-serif;
}

/* OPTION 2: Safer Fallback Stack */
body {
  font-family: "Inter", "Helvetica Neue", Arial, sans-serif;
}

/* OPTION 3: Serif Stack (elegant) */
body {
  font-family: "Georgia", "Times New Roman", serif;
}

/* OPTION 4: Monospace Stack (code) */
code {
  font-family: "Fira Code", "Cascadia Code", "Source Code Pro", 
               Menlo, "Courier New", monospace;
}

┌─────────────────────────────────────────────────────────────┐
│  RESPONSIVE TYPOGRAPHY SCALE (Harmonious Sizing)           │
└─────────────────────────────────────────────────────────────┘

:root {
  --text-xs: 0.75rem;   /* 12px */
  --text-sm: 0.875rem;  /* 14px */
  --text-base: 1rem;    /* 16px */
  --text-lg: 1.125rem;  /* 18px */
  --text-xl: 1.25rem;   /* 20px */
  --text-2xl: 1.5rem;   /* 24px */
  --text-3xl: 1.875rem; /* 30px */
  --text-4xl: 2.25rem;  /* 36px */
  --text-5xl: 3rem;     /* 48px */
}

h1 { font-size: var(--text-4xl); }
h2 { font-size: var(--text-2xl); }
p  { font-size: var(--text-base); }
small { font-size: var(--text-sm); }
